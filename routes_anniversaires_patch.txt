        if request.endpoint in (None, 'static') or request.path.startswith('/static/'):
            return None
        from anniversaires import notifier_anniversaires_du_jour
        notifier_anniversaires_du_jour()
    except Exception as e:
        print(f"[anniversaires] déclencheur indisponible : {e}")
    return None
